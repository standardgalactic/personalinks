The interpretation of this chain is as follows.

Distinctions generate information.

Repair preserves reachability.
Reachability determines admissibility.
Admissibility determines future possibility.
The geometry of future possibility is encoded in

\begin{proposition}[Hierarchy of Failure]{hierarchy-failure}
Failure at any level propagates upward through the
dependency chain.
\end{proposition}

\begin{proof}
If distinctions cannot be maintained, information cannot

cannot be reconstructed. Without recoverable histories,
repair becomes impossible. Without repair, reachable
volume contracts. As reachable volume contracts,
admissible volume contracts. Consequently admissibility
curvature becomes increasingly positive, signalling
collapse of future possibility.

Therefore degradation at any lower level necessarily
appears as admissibility loss at higher levels.
\end{proof}

Flow Computing


A flow system is a directed acyclic graph $G = (V,E)$
where vertices are transformations and edges transport
distinctions.
The computational state at time $t$ is
$X_t = \bigcup_{v \in V_t}\mathrm{Output}(v)$.

\begin{definition}[Distinction Transport Operator]{distinction-transport-operator}
Let

\[
G=(V,E)
\]

be a flow graph.

Define

\[
T_G :
\mathcal D
\rightarrow
\mathcal D
\]

by

\[
T_G(d)
=
\bigcup_{e\in E}
T_e(d)
\]

where

\[
T_e
\]

is the distinction transport induced by edge \(e\).
\end{definition}

\begin{theorem}[Flow Equivalence Theorem]{flow-equivalence}
Two flow systems

\[
G_1
\]

and

\[
G_2
\]

are observationally equivalent if and only if

\[
T_{G_1}
=
T_{G_2}.
\]
\end{theorem}

\begin{proof}
Assume

\[
T_{G_1}
=
T_{G_2}.
\]

For every initial distinction structure

\[
d
\]

both systems generate identical transported
distinctions.

Therefore every observable distinction available to an
observer is identical.

Hence observations coincide.

Conversely, suppose

\[
T_{G_1}
\neq
T_{G_2}.
\]

Then there exists

\[
d
\]

such that

\[
T_{G_1}(d)
\neq
T_{G_2}(d).
\]

Therefore at least one distinction differs between the
two systems.

An observer capable of accessing that distinction can
separate the systems.

Hence they are not observationally equivalent.
\end{proof}

Define transport entropy $S_F = -\sum_i p_i\log p_i$
where $p_i = f_i / \sum_j f_j$ and $f_i$ is edge flow.

\begin{theorem}{Flow Compression Theorem}{flow-compression}
  Maximum compression occurs when transport entropy is
  minimised subject to reachability constraints.
\end{theorem}

For composable flow networks, the transport operator
composes in the natural way.

\begin{definition}[Composable Flow Networks]{composable-flow}
  Flow networks $G_1 = (V_1, E_1)$ and $G_2 = (V_2, E_2)$
  are \emph{composable}, written $G_2 \circ G_1$, when the
  output vertices of $G_1$ coincide with the input vertices
  of $G_2$, so that the composite graph routes every output
  of $G_1$ into $G_2$.
\end{definition}

\begin{theorem}{Flow Composition Theorem}{flow-comp}
  For composable flow networks $G_1, G_2$, the induced
  transport operator satisfies
  \[
    T_{G_2 \circ G_1} = T_{G_2} \, T_{G_1}.
  \]
  Composition of transport operators is associative: for
  composable $G_1, G_2, G_3$,
  \[
    T_{G_3}\,(T_{G_2}\,T_{G_1})
    = (T_{G_3}\,T_{G_2})\,T_{G_1}.
  \]
\end{theorem}

\begin{proof}
  By the Flow Equivalence Theorem
  (\cref{thm:flow-equivalence}), the transport operator
  of a flow network is fully determined by its action on
  distinction structures. For $d \in \mathcal{D}$,
  $G_1$ first transports $d$ to $T_{G_1}(d)$, and since the
  output vertices of $G_1$ are exactly the input vertices of
  $G_2$ under composability, $G_2$ then transports this
  result to $T_{G_2}(T_{G_1}(d))$. By definition this is
  $T_{G_2 \circ G_1}(d)$, establishing the composition law.
  Associativity follows because function composition is
  associative on the underlying set $\mathcal{D}$: both
  sides of the associativity claim, applied to any $d$,
  reduce to the single expression
  $T_{G_3}(T_{G_2}(T_{G_1}(d)))$, since each side merely
  groups the same sequence of three applications differently.
\end{proof}

\begin{remark}{Why associativity matters}{flow-comp-assoc}
  Associativity guarantees that a flow pipeline's behaviour
  is independent of how it is decomposed into sub-pipelines
  for analysis or implementation. A long Unix pipeline can be
  split into stages and reasoned about piece by piece, in any
  grouping, without changing the transport operator the whole
  pipeline induces.
\end{remark}

Hence
\[
H_t
\sqsubseteq
H_{t+1}.
\]
\end{proof}

\begin{definition}{Refusal Witness}{refusal-witness}
  For a refusal $\mathbf{Refuse}(e)$ applied to
  configuration $(H, S \cup \{e\})$, the \emph{witness}
  is the record
  $w(e, H) = \langle e, |H|, \tau \rangle$ appended to
  history, where $|H|$ is the history length at the moment
  of refusal and $\tau$ is the refusal's timestamp.
\end{definition}

\begin{theorem}{Refusal Witness Theorem}{refusal-witness-thm}
  Every refusal generates a unique witness element in
  history: for distinct refusals
  $\mathbf{Refuse}(e)$ and $\mathbf{Refuse}(e')$
  occurring at different steps of the same execution,
  $w(e, H) \ne w(e', H')$.
\end{theorem}

\begin{proof}
  By the History Poset structure, $H$ strictly grows
  between distinct execution steps whenever a refusal
  occurs (the witness itself is appended), so any two
  refusals at different steps have witnesses with distinct
  $|H|$ components, hence distinct witnesses. Two refusals
  at the same step would require two simultaneous
  transitions from a single configuration, which the
  transition relation does not admit. Therefore witnesses
  are pairwise distinct, and each refusal's witness is
  unique to it.
\end{proof}

\begin{remark}{Why uniqueness of witnesses matters}{refusal-witness-rmk}
  The Refusal Witness Theorem is what makes
  $\mathbf{Refuse}$ genuinely optionality-preserving rather
  than a silent no-op. A silently dropped event would leave
  no trace distinguishing ``never offered'' from
  ``offered and declined'' in $H$, collapsing exactly the
  distinction Chapter~\ref{ch:spherepop}'s Refusal
  Preservation Theorem (\cref{thm:refusal-preserve}) depends
  on. The witness is what lets later repair or analysis
  recover that a genuine option existed and was declined,
  rather than never having existed at all.
\end{remark}

\begin{corollary}
Spherepop histories form a monotone chain in the history
lattice.
\end{corollary}


Collapse Geometry

\label{sec:app-collapse-geometry}

Define collapse quotient $Q_C = |S| / |\pi(S)|$.

\begin{theorem}{Collapse Bound}{collapse-bound}
  For finite scope, $Q_C \ge 1$.
\end{theorem}

\begin{proof}
  $\pi$ maps $S$ onto $\pi(S)$, so $|\pi(S)| \le |S|$.
  Since $|\pi(S)| \ge 1$ for nonempty $S$, the ratio
  $Q_C = |S|/|\pi(S)|$ is well-defined and at least $1$.
\end{proof}

\begin{definition}{Collapse Entropy}{collapse-entropy}
  The \emph{collapse entropy} of a collapse operation with
  quotient $Q_C$ is $S_C = \log Q_C$.
\end{definition}

\begin{prop}
  $S_C \ge 0$, with equality iff $\pi$ is injective on $S$
  (the collapse discards no distinctions).
\end{prop}

\begin{proof}
  Immediate from the Collapse Bound: $Q_C \ge 1$ implies
  $S_C = \log Q_C \ge 0$, with equality iff $Q_C = 1$ iff
  $|S| = |\pi(S)|$ iff $\pi$ is injective on $S$.
\end{proof}

\begin{theorem}{Collapse Entropy Monotonicity}{collapse-entropy-mono}
  Under successive collapses $\pi_1$ then $\pi_2$, total
  collapse entropy is additive and non-decreasing:
  \[
    S_C(\pi_2 \circ \pi_1) = S_C(\pi_1) + S_C(\pi_2|_{\pi_1(S)})
    \;\ge\; S_C(\pi_1).
  \]
\end{theorem}

\begin{proof}
  $Q_C(\pi_2 \circ \pi_1)
  = |S| / |\pi_2(\pi_1(S))|
  = (|S|/|\pi_1(S)|) \cdot (|\pi_1(S)|/|\pi_2(\pi_1(S))|)
  = Q_C(\pi_1) \cdot Q_C(\pi_2|_{\pi_1(S)})$.
  Taking logarithms gives the additivity claim. Since
  $S_C(\pi_2|_{\pi_1(S)}) \ge 0$ by the previous proposition,
  $S_C(\pi_2 \circ \pi_1) \ge S_C(\pi_1)$.
\end{proof}


HYDRA State Transitions


HYDRA state space: $M = \prod_{i=1}^k M_i$, with each
module state $M_i$ taking values in a common space — for
concreteness, $M_i \in \mathbb{R}^n$ — equipped with the
Euclidean metric $d(M_i, M_j) = \|M_i - M_j\|$.

Module dynamics: $\dot{M}_i = F_i(M)$.
Disagreement energy: $E_D = \sum_{i<j} d(M_i, M_j)^2$.

\begin{definition}{HYDRA Consensus Dynamics}{hydra-consensus-dyn}
  HYDRA's cross-module repair process realizes gradient
  descent on disagreement energy:
  \[
    F_i(M) = -\gamma \nabla_{M_i} E_D, \qquad \gamma > 0.
  \]
\end{definition}

\begin{prop}
  Under HYDRA Consensus Dynamics,
  $\nabla_{M_i} E_D = 2\sum_{j \ne i}(M_i - M_j)$, so
  explicitly
  \[
    \dot{M}_i = 2\gamma \sum_{j \ne i}(M_j - M_i).
  \]
\end{prop}

\begin{proof}
  $E_D = \sum_{i<j} \|M_i - M_j\|^2$. Each unordered pair
  $\{i,j\}$ with $j \ne i$ contributes exactly one term
  involving $M_i$, and differentiating $\|M_i-M_j\|^2$ with
  respect to $M_i$ gives $2(M_i - M_j)$. Summing over all
  $j \ne i$ gives $\nabla_{M_i}E_D = \sum_{j\ne i}2(M_i-M_j)$.
  Substituting into the consensus dynamics gives the stated
  form.
\end{proof}

This is the standard linear consensus rule: each module
moves toward the centroid of the others, at a rate
proportional to total disagreement.

\begin{theorem}{HYDRA Consensus Theorem}{hydra-conv}
  Under HYDRA Consensus Dynamics, $E_D$ is non-increasing
  along every trajectory,
  \[
    \dot{E}_D \le 0,
  \]
  with equality iff $M_1 = \cdots = M_k$. Consequently
  module consensus $M_i \to M^\star$ for all $i$ is globally
  asymptotically stable, where
  $M^\star = \frac{1}{k}\sum_i M_i(0)$.
\end{theorem}

\begin{proof}
  By the chain rule applied term by term to the defining
  sum,
  \[
    \dot{E}_D
    = \sum_{i<j} \frac{d}{dt} d(M_i, M_j)^2
    = \sum_{i<j} 2\, d(M_i, M_j) \, \frac{d}{dt} d(M_i, M_j).
  \]
  Equivalently, applying the multivariate chain rule
  directly to $E_D$ as a function of all module states,
  \[
    \dot{E}_D = \sum_{i=1}^k \nabla_{M_i} E_D \cdot \dot{M}_i.
  \]
  Substituting the consensus dynamics
  $\dot{M}_i = -\gamma \nabla_{M_i} E_D$,
  \[
    \dot{E}_D
    = \sum_i \nabla_{M_i}E_D \cdot (-\gamma \nabla_{M_i}E_D)
    = -\gamma \sum_i \|\nabla_{M_i}E_D\|^2
    \;\le\; 0.
  \]
  Equality holds iff $\nabla_{M_i}E_D = 0$ for every $i$,
  i.e.\ $\sum_{j\ne i}(M_i - M_j) = 0$ for every $i$. This
  condition rearranges to $kM_i = \sum_j M_j$ for every $i$,
  so every $M_i$ equals the same constant
  $\frac{1}{k}\sum_j M_j$; that is, $M_1 = \cdots = M_k$.
  Since $E_D \ge 0$ with $E_D = 0$ exactly at consensus,
  $E_D$ is a strict Lyapunov function away from consensus,
  and consensus is globally asymptotically stable by
  standard Lyapunov theory. The consensus value is the
  centroid: summing the dynamics over $i$,
  $\sum_i \dot{M}_i = 2\gamma \sum_i\sum_{j\ne i}(M_j - M_i)$,
  and every ordered pair $(M_j - M_i)$ in this double sum is
  cancelled by its reverse $(M_i - M_j)$, so
  $\sum_i \dot{M}_i = 0$: the centroid
  $\frac{1}{k}\sum_i M_i(0)$ is conserved and therefore
  equals the consensus value $M^\star$.
\end{proof}

\begin{remark}{Relation to the chapter's hand-off}{hydra-app-rmk}
  Chapter~\ref{ch:hydra}'s Hybrid Repair Theorem states the
  consensus condition abstractly, as
  $\dot{E}_D < 0$ away from equilibrium. This appendix shows
  that condition is not an independent assumption: under the
  natural gradient-flow model of cross-module repair,
  $\dot{E}_D \le 0$ is a derived consequence of the dynamics,
  not a hypothesis, with equality holding exactly at
  consensus.
\end{remark}


HYDRA Information Fusion


Beyond state consensus, HYDRA modules must also fuse their
probabilistic beliefs about the environment into a single
working posterior.

\begin{definition}{Module Posterior}{module-posterior}
  Each module $m_i$ maintains a posterior density $p_i(x)$
  over environment states $x$, reflecting its local
  distinction model $d_i$.
\end{definition}

\begin{definition}{Log-Opinion Pool Consensus}{log-opinion-pool}
  The HYDRA consensus posterior is the weighted
  log-opinion pool
  \[
    p_H(x) = \frac{1}{Z} \prod_i p_i(x)^{w_i},
    \qquad w_i \ge 0,\ \sum_i w_i = 1,
  \]
  with normalising constant
  $Z = \int \prod_i p_i(x)^{w_i}\,dx$.
\end{definition}

The general case admits no closed form, but the Gaussian
case — the setting in which HYDRA modules typically operate
when estimating continuous quantities such as projection
parameters — does, and gives a clean, fully derivable
information-gain result.

\begin{theorem}{Fusion Information Gain}{fusion-info-gain}
  Let modules maintain independent Gaussian posteriors
  $p_i(x) = \mathcal{N}(\mu_i, \sigma_i^2)$,
  $i = 1, \ldots, k$, over a shared scalar quantity $x$, and
  let $p_H$ be the log-opinion pool consensus with weights
  $w_i \ge 0$, $\sum_i w_i = 1$. Then $p_H$ is Gaussian with
  precision $\tau_H = \sum_i w_i \tau_i$, where
  $\tau_i = \sigma_i^{-2}$, and
  \[
    H(p_H) \;\le\; \max_i H(p_i),
  \]
  with equality iff every module assigned nonzero weight
  shares the same variance.
\end{theorem}

\begin{proof}
  For Gaussian $p_i$,
  $p_i(x) \propto \exp\bigl(-\tfrac{1}{2}\tau_i(x-\mu_i)^2\bigr)$,
  so
  \[
    \prod_i p_i(x)^{w_i}
    \;\propto\;
    \exp\!\Bigl(
      -\tfrac{1}{2} \sum_i w_i \tau_i (x - \mu_i)^2
    \Bigr).
  \]
  Expanding the exponent as a quadratic in $x$ and
  completing the square shows this is, up to a constant,
  $-\tfrac{1}{2}\tau_H (x - \mu_H)^2$ with
  $\tau_H = \sum_i w_i \tau_i$ and
  $\mu_H = \tau_H^{-1}\sum_i w_i \tau_i \mu_i$. Hence $p_H$
  is Gaussian with precision $\tau_H$, a weighted average of
  the individual precisions $\tau_i$ with weights $w_i$.
  Since $w_i \ge 0$ and $\sum_i w_i = 1$, any such weighted
  average satisfies
  $\min_i \tau_i \le \tau_H \le \max_i \tau_i$;
  in particular $\tau_H \ge \min_i \tau_i$, so
  $\sigma_H^2 = \tau_H^{-1} \le \max_i \sigma_i^2$.
  Gaussian differential entropy
  $H(\mathcal{N}(\mu,\sigma^2)) = \tfrac{1}{2}\log(2\pi e
  \sigma^2)$ is strictly increasing in $\sigma^2$, so
  $\sigma_H^2 \le \max_i \sigma_i^2$ gives
  $H(p_H) \le \max_i H(p_i)$ directly. Equality requires
  $\tau_H = \min_i \tau_i$ among weighted modules, which a
  weighted average achieves only when every weighted term
  equals that minimum, i.e.\ every module with $w_i>0$
  shares the same variance.
\end{proof}

\begin{remark}{Scope of the Gaussian case}{fusion-info-gain-rmk}
  Precision-weighting, $w_i \propto \tau_i$, is the natural
  choice — it lets more confident modules pull the consensus
  more strongly toward their own estimate — but it is not
  the unique variance-minimising weighting; maximising
  $\tau_H$ subject to $\sum_i w_i = 1$ is a linear objective
  over a simplex and is technically optimised by deferring
  entirely to the single highest-precision module, which is
  rarely the desired behaviour. HYDRA's weighting is chosen
  for robustness to single-module failure, not for
  worst-case variance minimisation. For non-Gaussian
  posteriors, the log-opinion pool need not be unimodal, and
  entropy comparisons of this kind require case-by-case
  analysis; the Gaussian case is presented here because it
  is both tractable and the setting HYDRA's continuous
  projection parameters typically occupy in practice.
\end{remark}


TARTAN Recursive Distinction


Let \(M\) be a semantic manifold equipped with a measure
\(\mu\). A TARTAN structure is a recursively refined
partition of \(M\). At depth \(0\), let

\[
\mathcal T_0=\{T_1,\ldots,T_m\}
\]

be an initial measurable partition. Recursive subdivision
produces partitions

\[
\mathcal T_0 \preceq \mathcal T_1 \preceq
\mathcal T_2 \preceq \cdots
\]

where each tile satisfies

\[
T_i=\bigcup_j T_{ij},
\qquad
T_{ij}\cap T_{ik}=\varnothing
\quad(j\neq k).
\]

The distinction capacity at depth \(n\) is

\[
D_n=\log |\mathcal T_n|.
\]

More generally, if the tiles carry weights \(p_i^{(n)}\),
the weighted distinction capacity is

\[
D_n
=
-\sum_{T_i\in\mathcal T_n}
p_i^{(n)}
\log p_i^{(n)}.
\]

\begin{definition}{Recursive Distinction Operator}{tartan-recursive-op}
  A recursive distinction operator is a map
  \[
    \mathcal R:\mathcal T_n\to \mathcal T_{n+1}
  \]
  such that every tile \(T\in\mathcal T_n\) is replaced by
  a finite measurable partition
  \[
    \mathcal R(T)=\{T_1,\ldots,T_{k(T)}\}
  \]
  with
  \[
    T=\bigcup_{\ell=1}^{k(T)}T_\ell,
    \qquad
    T_\ell\cap T_m=\varnothing
    \quad(\ell\neq m).
  \]
\end{definition}

\begin{definition}{Subdivision Factor}{tartan-subdivision-factor}
  The subdivision factor at depth \(n\) is
  \[
    \lambda_n
=
    \frac{|\mathcal T_{n+1}|}{|\mathcal T_n|}.
  \]
  If \(\lambda_n=\lambda\) for all \(n\), the TARTAN
  refinement is called homogeneous.
\end{definition}

\begin{theorem}{Recursive Distinction Theorem}{recursive-dist}
  If each subdivision increases distinguishability by a
  constant factor \(\lambda>1\), then after depth \(n\)
  the number of distinguishable tiles satisfies
  \[
    |\mathcal T_n|
=
    \lambda^n|\mathcal T_0|.
  \]
  Consequently the unweighted distinction capacity satisfies
  \[
    D_n
=
    D_0+n\log\lambda.
  \]
  If instead \(D_n\) denotes raw distinction count rather
  than logarithmic capacity, then
  \[
    D_n=\lambda^nD_0.
  \]
\end{theorem}

\begin{proof}
  By definition of homogeneous subdivision,
  \[
    |\mathcal T_{n+1}|
=
    \lambda|\mathcal T_n|.
  \]
  Iterating gives
  \[
    |\mathcal T_n|
=
    \lambda^n|\mathcal T_0|.
  \]
  The logarithmic distinction capacity is therefore
  \[
    D_n
=
    \log|\mathcal T_n|
=
    \log(\lambda^n|\mathcal T_0|)
=
    n\log\lambda+\log|\mathcal T_0|
=
    D_0+n\log\lambda.
  \]
  If \(D_n\) is used for the raw count
  \( |\mathcal T_n| \), the exponential formula follows
  directly.
\end{proof}

\begin{definition}{Distinction Density}{dist-density}
  Let \(\rho:M\to\mathbb R_{\ge0}\) denote local
  distinction density. For a measurable region \(U\subseteq M\),
  the local distinction mass is
  \[
    D(U)=\int_U \rho(x)\,d\mu(x).
  \]
\end{definition}

The semantic curvature associated with \(\rho\) is

\[
K_S
=
-\nabla^2\log\rho.
\]

In local coordinates this is the tensor

\[
(K_S)_{ij}
=
-\nabla_i\nabla_j\log\rho.
\]

Its trace is

\[
\operatorname{tr}(K_S)
=
-g^{ij}\nabla_i\nabla_j\log\rho
=
-\Delta\log\rho.
\]

Positive semantic curvature corresponds to local
concentration of distinctions and possible bottlenecking;
negative semantic curvature corresponds to local spreading
and semantic expansion.


Lyapunov Formulation


Let \(\rho(x,t)\) evolve by a refinement--diffusion
equation

\[
\frac{\partial \rho}{\partial t}
=
\alpha \Delta\rho
+
\beta R(\rho),
\]

where \(\alpha\ge0\) controls smoothing and \(\beta\ge0\)
controls recursive refinement. A simple stabilizing model
is

\[
R(\rho)=\rho\left(1-\frac{\rho}{\rho_*}\right),
\]

so that

\[
\frac{\partial \rho}{\partial t}
=
\alpha\Delta\rho
+
\beta\rho\left(1-\frac{\rho}{\rho_*}\right).
\]

Here \(\rho_*>0\) is the target distinction density.

Define the Lyapunov functional

\[
\mathcal L[\rho]
=
\frac12
\int_M
(\rho-\rho_*)^2
\,d\mu.
\]

Then

\[
\frac{d\mathcal L}{dt}
=
\int_M
(\rho-\rho_*)
\frac{\partial\rho}{\partial t}
\,d\mu.
\]

Substituting the dynamics gives

\[
\frac{d\mathcal L}{dt}
=
\alpha
\int_M
(\rho-\rho_*)\Delta\rho
\,d\mu
+
\beta
\int_M
(\rho-\rho_*)
\rho\left(1-\frac{\rho}{\rho_*}\right)
\,d\mu.
\]

Assume either compact \(M\) without boundary or boundary
conditions eliminating boundary terms. Integration by parts
yields

\[
\int_M
(\rho-\rho_*)\Delta\rho
\,d\mu
=
-\int_M
\|\nabla\rho\|^2
\,d\mu.
\]

For the reaction term, observe that

\[
\rho\left(1-\frac{\rho}{\rho_*}\right)
=
\frac{\rho}{\rho_*}(\rho_*-\rho)
=
-\frac{\rho}{\rho_*}(\rho-\rho_*).
\]

Hence

\[
(\rho-\rho_*)
\rho\left(1-\frac{\rho}{\rho_*}\right)
=
-\frac{\rho}{\rho_*}(\rho-\rho_*)^2.
\]

Therefore

\[
\frac{d\mathcal L}{dt}
=
-\alpha
\int_M
\|\nabla\rho\|^2
\,d\mu
-
\beta
\int_M
\frac{\rho}{\rho_*}
(\rho-\rho_*)^2
\,d\mu.
\]

Since \(\alpha,\beta,\rho,\rho_*\ge0\), both terms are
nonpositive:

\[
\frac{d\mathcal L}{dt}\le0.
\]

\begin{theorem}{TARTAN Lyapunov Stability}{tartan-lyapunov}
  Under compact or no-flux boundary conditions, the target
  distinction density \(\rho_*\) is Lyapunov stable for
  the refinement--diffusion dynamics
  \[
    \frac{\partial \rho}{\partial t}
=
    \alpha\Delta\rho
    +
    \beta\rho\left(1-\frac{\rho}{\rho_*}\right).
  \]
\end{theorem}

\begin{proof}
  The functional
  \[
    \mathcal L[\rho]
=
    \frac12
    \int_M
    (\rho-\rho_*)^2
    \,d\mu
  \]
  satisfies \(\mathcal L[\rho]\ge0\), with equality iff
  \(\rho=\rho_*\) almost everywhere. The calculation above
  gives
  \[
    \frac{d\mathcal L}{dt}
=
    -\alpha
    \int_M
    \|\nabla\rho\|^2
    \,d\mu
-
    \beta
    \int_M
    \frac{\rho}{\rho_*}
    (\rho-\rho_*)^2
    \,d\mu
    \le0.
  \]
  Hence \(\mathcal L\) is nonincreasing along trajectories.
  Therefore \(\rho_*\) is Lyapunov stable.
\end{proof}

If \(\alpha>0\), \(\beta>0\), and \(\rho>0\) almost
everywhere, then equality

\[
\frac{d\mathcal L}{dt}=0
\]

requires

\[
\nabla\rho=0
\]

and

\[
\rho=\rho_*.
\]

Thus the equilibrium is asymptotically stable modulo
measure-zero degeneracies.

\begin{corollary}{Asymptotic Stabilization}{tartan-asymptotic}
  If \(\alpha>0\), \(\beta>0\), and \(\rho(x,t)>0\) almost
  everywhere, then \(\rho(t)\to\rho_*\) in \(L^2(M)\).
\end{corollary}

\begin{proof}
  The Lyapunov functional is bounded below and
  nonincreasing, hence convergent. Its derivative vanishes
  only at the invariant set \(\rho=\rho_*\). By LaSalle's
  invariance principle, trajectories converge to that set.
\end{proof}


MEM|8 Memory Architecture


Let \(X\) be a cue space, event space, or semantic state
space equipped with a measure \(\mu\). Stored events are
represented as points or structured elements

\[
e_i\in X,
\qquad
i=1,\ldots,N.
\]

Each event carries a weight

\[
w_i\ge0,
\]

representing salience, reinforcement, recency, frequency,
or repair relevance.

Let

\[
K:X\times X\rightarrow\mathbb R_{\ge0}
\]

be a similarity kernel. The memory field generated by the
stored events is

\[
M(x,t)
=
\sum_{i=1}^{N}
w_i(t)K(x,e_i).
\]

The value \(M(x,t)\) measures the activation of stored
events under cue \(x\) at time \(t\).

\begin{definition}{Memory Kernel}{mem-kernel}
  A memory kernel is a function
  \[
    K:X\times X\to\mathbb R_{\ge0}
  \]
  satisfying
  \[
    K(x,e)\ge0
  \]
  and usually
  \[
    K(x,e)=K(e,x).
  \]
  If \(K\) is positive definite, then for any finite set
  \(\{x_1,\ldots,x_N\}\subset X\) and coefficients
  \(a_i\in\mathbb R\),
  \[
    \sum_{i,j=1}^N a_i a_j K(x_i,x_j)\ge0.
  \]
\end{definition}

A common example is the Gaussian kernel

\[
K(x,e)
=
\exp
\left(
-\frac{d(x,e)^2}{2\sigma^2}
\right),
\]

where \(\sigma>0\) controls the radius of association.

Small \(\sigma\) produces sharply localized recall.

Large \(\sigma\) produces broad associative recall.

\begin{definition}{Memory Field}{mem-field}
  Given events \(e_i\), weights \(w_i(t)\), and kernel
  \(K\), the memory field is
  \[
    M(x,t)
=
    \sum_{i=1}^{N}
    w_i(t)K(x,e_i).
  \]
\end{definition}


Ecphory


Ecphory is the activation of a stored memory by a cue. In
this formalization, recall occurs when the memory field
crosses a threshold.

Let

\[
\theta>0
\]

be a recall threshold.

\begin{definition}{Ecphoric Region}{ecphoric-region}
  The ecphoric region at time \(t\) is
  \[
    E_\theta(t)
=
    \{x\in X:M(x,t)>\theta\}.
  \]
\end{definition}

\begin{definition}{Recall Indicator}{recall-indicator}
  The recall indicator is
  \[
    \chi_\theta(x,t)
=
    \begin{cases}
      1, & M(x,t)>\theta,\\
      0, & M(x,t)\le\theta.
    \end{cases}
  \]
\end{definition}

\begin{theorem}{Ecphory Threshold Theorem}{ecphory}
  Recall occurs at cue \(x\) and time \(t\) if and only if
  \[
    \sum_{i=1}^{N} w_i(t)K(x,e_i)>\theta.
  \]
\end{theorem}

\begin{proof}
  By definition, recall occurs iff
  \[
    M(x,t)>\theta.
  \]
  Substituting the definition of \(M\) gives
  \[
    \sum_{i=1}^{N} w_i(t)K(x,e_i)>\theta.
  \]
  Hence the two conditions are equivalent.
\end{proof}


Single-Event Recall Radius


For a single stored event \(e\) with weight \(w\), the
memory field is

\[
M(x)
=
wK(x,e).
\]

For the Gaussian kernel,

\[
M(x)
=
w
\exp
\left(
-\frac{d(x,e)^2}{2\sigma^2}
\right).
\]

Recall occurs iff

\[
w
\exp
\left(
-\frac{d(x,e)^2}{2\sigma^2}
\right)
>
\theta.
\]

Assuming \(w>\theta\), taking logarithms yields

\[
-\frac{d(x,e)^2}{2\sigma^2}
>
\log\left(\frac{\theta}{w}\right).
\]

Thus

\[
d(x,e)^2
<
2\sigma^2
\log\left(\frac{w}{\theta}\right).
\]

\begin{proposition}{Gaussian Recall Radius}{gaussian-recall-radius}
  For a single Gaussian memory trace, recall occurs inside
  the ball
  \[
    d(x,e)
    <
    r_\theta,
  \]
  where
  \[
    r_\theta
=
    \sigma
    \sqrt{
      2\log\left(\frac{w}{\theta}\right)
    }.
  \]
  If \(w\le\theta\), no cue recalls the event.
\end{proposition}

\begin{proof}
  The inequality above gives the result directly. If
  \(w\le\theta\), the maximum of the field occurs at
  \(x=e\), where \(M(e)=w\le\theta\), so the threshold is
  never crossed.
\end{proof}


Superposition and Associative Recall


For multiple events, recall can occur even if no single
trace individually crosses threshold.

\begin{proposition}{Associative Summation}{assoc-summation}
  Suppose
  \[
    w_iK(x,e_i)\le\theta
  \]
  for every \(i\), but
  \[
    \sum_{i=1}^N w_iK(x,e_i)>\theta.
  \]
  Then recall occurs by associative summation.
\end{proposition}

\begin{proof}
  Recall depends on the total field
  \[
    M(x)=\sum_i w_iK(x,e_i),
  \]
  not on any single summand. Therefore the total may cross
  threshold even when no individual term does.
\end{proof}

This formalizes pattern completion. A cue need not match
one stored event exactly. It may activate a cluster of
nearby traces whose combined field exceeds threshold.


Interference


If many events have overlapping kernels, recall becomes
ambiguous. Define the contribution of event \(e_i\) at cue
\(x\) by

\[
a_i(x,t)
=
w_i(t)K(x,e_i).
\]

The normalized attribution is

\[
\alpha_i(x,t)
=
\frac{a_i(x,t)}
{\sum_j a_j(x,t)}
\]

whenever \(M(x,t)>0\).

\begin{definition}{Recall Ambiguity}{recall-ambiguity}
  Recall ambiguity is the entropy
  \[
    H_{\mathrm{rec}}(x,t)
=
-
    \sum_i
    \alpha_i(x,t)\log\alpha_i(x,t).
  \]
\end{definition}

Low ambiguity means one trace dominates recall.

High ambiguity means many traces contribute comparably.

\begin{proposition}{Dominant Trace Criterion}{dominant-trace}
  If
  \[
    \alpha_k(x,t)>\frac12,
  \]
  then event \(e_k\) contributes more activation than all
  other events combined.
\end{proposition}

\begin{proof}
  Since
  \[
    \sum_i\alpha_i=1,
  \]
  the contribution of all events other than \(k\) is
  \[
    1-\alpha_k.
  \]
  If \(\alpha_k>1/2\), then
  \[
    \alpha_k>1-\alpha_k.
  \]
\end{proof}


Memory Decay


Let memory weights decay according to

\[
\frac{dw_i}{dt}
=
-\lambda_i w_i,
\qquad
\lambda_i\ge0.
\]

Then

\[
w_i(t)
=
w_i(0)e^{-\lambda_i t}.
\]

The memory field becomes

\[
M(x,t)
=
\sum_i
w_i(0)e^{-\lambda_i t}K(x,e_i).
\]

\begin{theorem}{Exponential Forgetting}{exp-forgetting}
  If all decay rates satisfy
  \[
    \lambda_i\ge\lambda_{\min}>0,
  \]
  then
  \[
    M(x,t)
    \le
    e^{-\lambda_{\min}t}M(x,0).
  \]
\end{theorem}

\begin{proof}
  Since
  \[
    e^{-\lambda_i t}\le e^{-\lambda_{\min}t}
  \]
  for all \(i\), and \(w_iK(x,e_i)\ge0\),
  \[
    M(x,t)
=
    \sum_i
    w_i(0)e^{-\lambda_i t}K(x,e_i)
    \le
    e^{-\lambda_{\min}t}
    \sum_i w_i(0)K(x,e_i).
  \]
  The final sum is \(M(x,0)\).
\end{proof}

\begin{corollary}{Recall Time Bound}{recall-time-bound}
  If
  \[
    M(x,0)>\theta
  \]
  and all traces decay with common rate \(\lambda\), then
  recall at \(x\) persists until
  \[
    t
    <
    \frac1\lambda
    \log
    \left(
      \frac{M(x,0)}{\theta}
    \right).
  \]
\end{corollary}

\begin{proof}
  With common decay rate,
  \[
    M(x,t)=e^{-\lambda t}M(x,0).
  \]
  Recall requires
  \[
    e^{-\lambda t}M(x,0)>\theta.
  \]
  Solving gives
  \[
    t<
    \frac1\lambda
    \log
    \left(
      \frac{M(x,0)}{\theta}
    \right).
  \]
\end{proof}


Reinforcement


Memory traces may also be reinforced. Let cue exposure add
weight according to

\[
\frac{dw_i}{dt}
=
-\lambda_i w_i
+
\eta_i r_i(t),
\]

where \(r_i(t)\ge0\) is reinforcement signal and
\(\eta_i\ge0\) is learning rate.

For constant reinforcement \(r_i(t)=r_i\), the solution is

\[
w_i(t)
=
w_i(0)e^{-\lambda_i t}
+
\frac{\eta_i r_i}{\lambda_i}
\left(1-e^{-\lambda_i t}\right).
\]

Thus

\[
\lim_{t\to\infty}w_i(t)
=
\frac{\eta_i r_i}{\lambda_i}.
\]

\begin{proposition}{Reinforced Memory Equilibrium}{reinforced-memory-equilibrium}
  Under constant reinforcement, the equilibrium trace
  weight is
  \[
    w_i^*
=
    \frac{\eta_i r_i}{\lambda_i}.
  \]
\end{proposition}

\begin{proof}
  At equilibrium,
  \[
    \frac{dw_i}{dt}=0.
  \]
  Therefore
  \[
    -\lambda_i w_i+\eta_i r_i=0,
  \]
  and hence
  \[
    w_i^*=\frac{\eta_i r_i}{\lambda_i}.
  \]
\end{proof}


Kernel Memory as Projection


Let

\[
\Phi:X\to\mathcal H_K
\]

be the feature map into the reproducing-kernel Hilbert
space associated with \(K\), so that

\[
K(x,e)
=
\langle \Phi(x),\Phi(e)\rangle_{\mathcal H_K}.
\]

Then

\[
M(x,t)
=
\left\langle
\Phi(x),
\sum_i w_i(t)\Phi(e_i)
\right\rangle_{\mathcal H_K}.
\]

Define the memory vector

\[
m(t)
=
\sum_i w_i(t)\Phi(e_i).
\]

Then

\[
M(x,t)
=
\langle\Phi(x),m(t)\rangle_{\mathcal H_K}.
\]

\begin{proposition}{Kernel Representation of Memory}{kernel-memory-representation}
  The MEM\(|\)8 memory field is a linear functional in
  feature space.
\end{proposition}

\begin{proof}
  The result follows by substituting the reproducing
  property
  \[
    K(x,e_i)
=
    \langle\Phi(x),\Phi(e_i)\rangle
  \]
  into the definition of \(M(x,t)\).
\end{proof}

Thus memory recall is nonlinear in original cue space but
linear in kernel feature space.


False Recall


False recall occurs when

\[
M(x,t)>\theta
\]

for a cue \(x\) not corresponding to any intended stored
event.

Let

\[
B_i(r)
=
\{x:d(x,e_i)<r\}
\]

denote the intended recall neighborhood of \(e_i\).

Define the intended region

\[
B(r)
=
\bigcup_i B_i(r).
\]

\begin{definition}{False Recall Region}{false-recall-region}
  The false recall region is
  \[
    F_\theta(t)
=
    E_\theta(t)\setminus B(r).
  \]
\end{definition}

Its measure is

\[
\mu(F_\theta(t)).
\]

\begin{definition}{False Recall Rate}{false-recall-rate}
  The false recall rate is
  \[
    \mathrm{FR}(t)
=
    \frac{\mu(F_\theta(t))}
    {\mu(E_\theta(t))}
  \]
  whenever \(\mu(E_\theta(t))>0\).
\end{definition}

This gives a geometric measure of hallucinated recall,
confabulation, or spurious associative activation.


Memory Capacity Bound


Suppose the cue space has finite measure

\[
\mu(X)<\infty.
\]

Assume each event has a recall neighborhood of radius
\(r_\theta\) and that these neighborhoods must remain
disjoint for unambiguous recall.

If each ball has measure

\[
b_\theta=\mu(B_i(r_\theta)),
\]

then the maximum number of noninterfering memories is
bounded by

\[
N_{\max}
\le
\frac{\mu(X)}{b_\theta}.
\]

\begin{theorem}{Noninterference Capacity Bound}{noninterference-capacity}
  If unambiguous recall requires pairwise disjoint
  ecphoric neighborhoods, then
  \[
    N
    \le
    \frac{\mu(X)}{b_\theta}.
  \]
\end{theorem}

\begin{proof}
  If recall neighborhoods are disjoint, then
  \[
    \mu\left(\bigcup_{i=1}^N B_i(r_\theta)\right)
=
    \sum_{i=1}^N \mu(B_i(r_\theta))
=
    Nb_\theta.
  \]
  Since the union is contained in \(X\),
  \[
    Nb_\theta\le\mu(X).
  \]
  Therefore
  \[
    N\le\frac{\mu(X)}{b_\theta}.
  \]
\end{proof}


Repair Interpretation


A memory trace is recoverable when an appropriate cue can
reconstruct it. Thus recoverability may be written as

\[
\reco(e_i)
=
\mu
\left(
\{x:w_iK(x,e_i)>\theta\}
\right).
\]

For a Gaussian trace with \(w_i>\theta\),

\[
\reco(e_i)
=
\mu(B_i(r_\theta)).
\]

Hence recoverability is proportional to the measure of the
ecphoric basin.

\begin{proposition}{Recoverability--Basin Relation}{recoverability-basin}
  For single-trace Gaussian memory, recoverability
  increases monotonically with
  \[
    \sigma
    \sqrt{
      2\log\left(\frac{w_i}{\theta}\right)
    }.
  \]
\end{proposition}

\begin{proof}
  The ecphoric basin radius is
  \[
    r_\theta
=
    \sigma
    \sqrt{
      2\log\left(\frac{w_i}{\theta}\right)
    }.
  \]
  In ordinary metric spaces, ball measure is monotone in
  radius. Therefore recoverability increases monotonically
  with \(r_\theta\).
\end{proof}
